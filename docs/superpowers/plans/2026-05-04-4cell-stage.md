```python
def divide_cell_body(
    parent,
    positions,
    axis=None,
    volume_ratio=0.5,
    area_threshold=None,
    daughter_names=(None, None),
    daughter_orientations=(None, None),
):
    """
    Divide a CellBody into two daughters.

    Args:
        parent: CellBody to divide
        positions: (N, 3) particle positions array
        axis: division axis (unit vector). Random if None.
        volume_ratio: fraction of volume for first daughter (0.0-1.0)
        area_threshold: if set, passed to create_cell_body for adaptive subdivision
        daughter_names: (name1, name2) biological names for daughters
        daughter_orientations: (orient1, orient2) 3x3 orientation matrices.
            If None for either, inherits parent orientation.
    """
    if axis is None:
        axis = np.random.randn(3)
    axis = axis / np.linalg.norm(axis)

    parent_center = positions[parent.center_idx].copy()
    parent_volume = parent.target_volume

    parent_radius = (3.0 * parent_volume / 4.0) ** (1.0 / 3.0)

    r1 = parent_radius * (volume_ratio ** (1.0 / 3.0))
    r2 = parent_radius * ((1.0 - volume_ratio) ** (1.0 / 3.0))

    offset1 = 0.6 * r1
    offset2 = 0.6 * r2
    center1 = parent_center + offset1 * axis
    center2 = parent_center - offset2 * axis

    parent_indices = sorted(parent.all_indices(), reverse=True)
    positions = np.delete(positions, parent_indices, axis=0)

    orient1 = daughter_orientations[0] if daughter_orientations[0] is not None else parent.orientation.copy()
    orient2 = daughter_orientations[1] if daughter_orientations[1] is not None else parent.orientation.copy()

    d1, positions = create_cell_body(
        center=center1,
        radius=r1,
        positions=positions,
        membrane_compliance=parent.membrane_compliance,
        area_threshold=area_threshold,
        name=daughter_names[0],
        orientation=orient1,
    )
    d2, positions = create_cell_body(
        center=center2,
        radius=r2,
        positions=positions,
        membrane_compliance=parent.membrane_compliance,
        area_threshold=area_threshold,
        name=daughter_names[1],
        orientation=orient2,
    )

    return [d1, d2], positions
```

def _apply_lineage_divisions(self):
    program = self.lineage.get_step_program(self.step_count)
    if program is None:
        return

    for rule in program.divisions:
        idx = next(
            (i for i, cb in enumerate(self.cell_bodies) if cb.name == rule.parent),
            None,
        )
        if idx is None:
            continue

        parent = self.cell_bodies[idx]
        axis = self.lineage.resolve_division_axis(rule, parent.orientation)
        d_orientations = (parent.orientation.copy(), parent.orientation.copy())
        daughters, self.positions = divide_cell_body(
            parent,
            self.positions,
            axis=axis,
            volume_ratio=rule.volume_ratio,
            daughter_names=tuple(rule.daughters),
            daughter_orientations=d_orientations,
        )

        self.cell_bodies.pop(idx)
        self.cell_bodies.extend(daughters)

def step(self):
    # Events
    if self.lineage is not None:
        self._apply_lineage_divisions()
    elif self.step_count in self.division_steps:
        idx = max(
            range(len(self.cell_bodies)),
            key=lambda i: self.cell_bodies[i].target_volume,
        )
        parent = self.cell_bodies[idx]

        daughters, self.positions = divide_cell_body(parent, self.positions)

        self.cell_bodies.pop(idx)
        self.cell_bodies.extend(daughters)

        self.positions, self.inv_masses = build_particle_arrays(
            self.cell_bodies, self.positions
        )
        self.velocities = np.zeros_like(self.positions)
        self.active_adhesions.clear()

        centers = np.array(
            [self.positions[cb.center_idx] for cb in self.cell_bodies]
        )
        self.cell_polarity = self.polarity.value(centers)

    # Physics
    self.positions, self.velocities = step(
        self.positions,
        self.velocities,
        self.inv_masses,
        self.dt,
        self.cell_bodies,
        self.shell_params,
        active_adhesions=self.active_adhesions,
    )

    centers = np.array([self.positions[cb.center_idx] for cb in self.cell_bodies])
    self.cell_polarity = self.polarity.value(centers)

    self.step_count += 1
    self.time += self.dt
```

"""
Demo: C. elegans 4-cell stage via XPBD simulation.

Runs lineage-driven division from 1-cell (P0) to 4-cell stage (ABa, ABp, EMS, P2),
then lets the XPBD solver settle cells into a physical arrangement.

Run with:
    uv run python scripts/demo_4cell_stage.py
"""

import numpy as np
import matplotlib.pyplot as plt
from mpl_toolkits.mplot3d import Axes3D  # noqa: F401

from develmorph.cell_body import create_cell_body
from develmorph.celllineage import CellLineage
from develmorph.engine import World

YAML_PATH = "develmorph/lineages/celegans_early.yaml"
CELL_COLORS = {
    "ABa": "royalblue",
    "ABp": "cornflowerblue",
    "EMS": "tomato",
    "P2":  "darkorange",
}


def run_simulation(n_division_steps=4, n_settle_steps=200):
    lineage = CellLineage(YAML_PATH)

    positions = np.zeros((0, 3))
    p0, positions = create_cell_body(
        center=np.array([0.0, 0.0, 0.0]),
        radius=1.0,
        positions=positions,
        name="P0",
    )

    world = World(
        cell_bodies=[p0],
        positions=positions,
        lineage=lineage,
        shell_params=(4.0, 4.0, 4.0),
        dt=0.01,
    )

    # Division phase: steps 0-3 (division at step_counts 1, 2, 3)
    for _ in range(n_division_steps):
        world.step()

    print(f"After {n_division_steps} steps: {[cb.name for cb in world.cell_bodies]}")

    # Settling phase: continue stepping (no more divisions until step_count 4)
    for _ in range(n_settle_steps):
        world.step()

    return world


def plot_4cell(world):
    fig = plt.figure(figsize=(8, 6))
    ax = fig.add_subplot(111, projection="3d")

    for cb in world.cell_bodies:
        center = world.positions[cb.center_idx]
        color = CELL_COLORS.get(cb.name, "gray")
        r = (3.0 * cb.target_volume / 4.0) ** (1.0 / 3.0)

        u = np.linspace(0, 2 * np.pi, 20)
        v = np.linspace(0, np.pi, 20)
        x = center[0] + r * np.outer(np.cos(u), np.sin(v))
        y = center[1] + r * np.outer(np.sin(u), np.sin(v))
        z = center[2] + r * np.outer(np.ones_like(u), np.cos(v))
        ax.plot_surface(x, y, z, alpha=0.4, color=color)
        ax.text(*center, cb.name, fontsize=9, ha="center")

    ax.set_xlim(-3, 3)
    ax.set_ylim(-3, 3)
    ax.set_zlim(-3, 3)
    ax.set_xlabel("X (A-P axis)")
    ax.set_ylabel("Y")
    ax.set_zlabel("Z")
    ax.set_title("C. elegans 4-cell stage (XPBD)")
    plt.tight_layout()
    plt.show()


if __name__ == "__main__":
    world = run_simulation()
    plot_4cell(world)
```
